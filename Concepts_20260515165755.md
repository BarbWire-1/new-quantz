// Diese internen Objekte bleiben die Standard-Datenspeicher
const RENDER_STATS = { instanceBindings: 0, attrBound: 0, exprRuns: 0 };
const TimeLog = { starts: new Map() };
const TimeAgg = { total: new Map(), count: new Map() };

// DER KERN: Ein globales Interface für jedes zukünftige Debug-Tool/Plugin
if (typeof window !== 'undefined' && !window.__QUANTZ_DEVTOOLS__) {
	window.__QUANTZ_DEVTOOLS__ = {
		stats: RENDER_STATS,
		timeAgg: TimeAgg,
		// Ermöglicht es dem Plugin, eigene, neue Messbereiche hinzuzufügen!
		trackCustomMetric: (key, value) => {
			RENDER_STATS[key] = (RENDER_STATS[key] || 0) + value;
		},
		getReport: () => {
			const rows = [...TimeAgg.total.entries()].map(([label, total]) => {
				const calls = TimeAgg.count.get(label) || 1;
				return {
					Task: label,
					'Total (ms)': total.toFixed(2),
					Calls: calls,
					'Avg (ms)': (total / calls).toFixed(3),
				};
			});
			return { summaryTable: rows, stats: RENDER_STATS };
		}
	};
}

// Ihre bestehenden Zeitmess-Funktionen füttern einfach dieses globale System
function trace(type, ...args) {
	if (import.meta.env.DEV) {
		console.log(`%c[${type.toUpperCase()}]`, 'color: cyan; font-weight: bold;', ...args);
	}
}

function timeStart(label) {
	if (!TimeLog.starts.has(label)) TimeLog.starts.set(label, []);
	TimeLog.starts.get(label).push(performance.now());
}

function timeEnd(label, meta = '') {
	const stack = TimeLog.starts.get(label);
	if (!stack?.length) return;
	const start = stack.pop();
	const ms = performance.now() - start;
	const full = meta ? `${label} ${meta}` : label;

	TimeAgg.total.set(full, (TimeAgg.total.set(full) || 0) + ms);
	TimeAgg.count.set(full, (TimeAgg.count.set(full) || 0) + 1);
}

function timed(label, meta, fn) {
	if (typeof meta === 'function') { fn = meta; meta = ''; }
	timeStart(label);
	const res = fn();
	timeEnd(label, meta);
	return res;
}

export { timeStart, timeEnd, timed, trace, RENDER_STATS };

export const QuantzDebugTool = {
	// Zeigt den aktuellen Report an
	printReport() {
		if (typeof window === 'undefined' || !window.__QUANTZ_DEVTOOLS__) {
			console.warn('Quantz Debug Tool läuft nicht im Entwicklungsmodus.');
			return;
		}

		const { summaryTable, stats } = window.__QUANTZ_DEVTOOLS__.getReport();

		console.log('%c📊 QUANTZ EXTENDED PERFORMANCE REPORT', 'color: magenta; font-weight: bold; font-size: 14px;');
		console.table(summaryTable);
		console.log('Engine & Custom Stats:', stats);
	},

	// HIER IST DIE ANTWORT FÜR ANDERE BEREICHE:
	// Erlaubt es dem User oder anderen Plugins, eigene Messungen einzuspeisen!
	trackUserProcess(label, fn) {
		if (typeof window !== 'undefined' && window.__QUANTZ_DEVTOOLS__) {
			const start = performance.now();
			const result = fn();
			const duration = performance.now() - start;

			// Wir pushen die Custom-Zeit direkt in das zentrale Engine-System!
			const totalMap = window.__QUANTZ_DEVTOOLS__.timeAgg.total;
			const countMap = window.__QUANTZ_DEVTOOLS__.timeAgg.count;

			totalMap.set(label, (totalMap.get(label) || 0) + duration);
			countMap.set(label, (countMap.get(label) || 0) + 1);

			return result;
		}
		return fn();
	}
};
